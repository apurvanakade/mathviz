ODE with a convergence plot

Euler and RK4 on the same problem, and the order read off a log-log fit

A Plotly chart of the numerical solution against the exact one, and beneath it an Observable Plot chart of error against step size on log axes – two libraries, one theme. The slope of the fitted line is the observed order of the method.

VM = window.VM
vmTheme = Generators.observe(notify => VM.plotting.onThemeChange(notify))
chartColors = VM.plotting.colors(vmTheme)

Inputs

The exact solution is typed too, so the page can draw the error. (For y' = y − t² + 1, y(0) = 0.5 it is (t + 1)² − e^t / 2.)

viewof odeRhs = Inputs.text({label: "y′ = f(t, y)", value: "y - t^2 + 1"})
viewof odeExact = Inputs.text({label: "exact y(t)", value: "(t + 1)^2 - e^t / 2"})
viewof odeY0 = Inputs.text({label: "y(0)", value: "0.5"})
viewof odeMethod = Inputs.radio(["Euler", "RK4"], {label: "method", value: "RK4"})

Solving

makeFunction2 compiles the right-hand side as (t, y) => number; makeFunctionOfT compiles the exact solution in t. The two steppers share a signature, so the method choice is a function reference.

ode = {
  const f = VM.expressions.makeFunction2(math, odeRhs)
  const exact = VM.expressions.makeFunctionOfT(math, odeExact)
  const y0 = VM.expressions.makeNumber(math, odeY0)
  if (f === null || exact === null || y0 === null) return null
  const solve = odeMethod === "Euler" ? VM.numerical.eulerSolve : VM.numerical.rk4Solve
  return {f, exact, y0, solve}
}

The solution

viewof odeN = Inputs.range([2, 64], {label: "steps n", step: 1, value: 8})
{
  const data = []
  if (ode !== null) {
    const sol = ode.solve(ode.f, 0, ode.y0, 2, odeN)
    const et = [], ey = []
    for (let i = 0; i <= 200; i++) { const t = 2 * i / 200; et.push(t); ey.push(ode.exact(t)) }
    data.push({x: et, y: ey, name: "exact", line: {color: chartColors.muted, width: 2, dash: "dot"}})
    data.push({x: sol.ts, y: sol.ys, name: odeMethod, mode: "lines+markers", line: {color: chartColors.fn, width: 2}, marker: {size: 6, line: {color: chartColors.halo, width: 1.5}}})
  }
  const div = document.createElement("div")
  div.style.height = "360px"
  Plotly.newPlot(div, data, VM.plotting.layout({
    xaxis: {title: "t"}, yaxis: {title: "y"},
    legend: {x: 0.02, y: 0.98},
    annotations: ode ? [] : VM.plotting.emptyState("Check the inputs")
  }), VM.plotting.config())
  VM.plotting.autoResize(div)
  return div
}

The convergence plot

Run the solver at n = 4, 8, 16, …, record the error at t = 2, fit a line to (log h, log error). linearRegression returns the slope – the order – and xlo/xhi (the first and last point’s x, in the order given) for drawing the fitted segment.

convergence = {
  if (ode === null) return null
  const points = []
  for (let n = 4; n <= 1024; n *= 2) {
    const sol = ode.solve(ode.f, 0, ode.y0, 2, n)
    const err = Math.abs(sol.ys[n] - ode.exact(2))
    if (err > 0 && Number.isFinite(err)) points.push({x: Math.log10(2 / n), y: Math.log10(err), h: 2 / n, err})
  }
  return {points, fit: VM.numerical.linearRegression(points)}
}
{
  if (convergence === null || convergence.fit === null) return html`<em>Not enough data for a fit.</em>`
  const {points, fit} = convergence
  const line = [{x: fit.xlo, y: fit.slope * fit.xlo + fit.intercept}, {x: fit.xhi, y: fit.slope * fit.xhi + fit.intercept}]
  return Plot.plot(VM.plotting.plotOptions({
    height: 300,
    x: {label: "log₁₀ h"}, y: {label: "log₁₀ |error at t = 2|"},
    marks: [
      Plot.line(line, {x: "x", y: "y", stroke: chartColors.alt, strokeWidth: 1.5, strokeDasharray: "4 3"}),
      Plot.dot(points, {x: "x", y: "y", fill: chartColors.fn, r: 4, stroke: chartColors.halo}),
      Plot.text([{x: fit.xhi, y: fit.slope * fit.xhi + fit.intercept}], {x: "x", y: "y", text: () => `slope ≈ ${fit.slope.toFixed(2)}`, dx: 8, dy: -10, textAnchor: "end", fill: chartColors.ink})
    ]
  }))
}

Euler’s slope sits near 1, RK4’s near 4 – until the error floors out in floating point at the smallest steps, which drags the fitted slope down. l1Regression would shrug off those last points; that swap is one identifier.
